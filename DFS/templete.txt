# Graph DFS 
def dfs(node, visited, graph):
    
    if node in visited:
        return
    
    visited.add(node)
    
    for neighbor in graph[node]:
        dfs(neighbor, visited, graph)


# Tree DFS 
def dfs(root):

    if not root:
        return

    # process node
    print(root.val)

    dfs(root.left)
    dfs(root.right)
